- `basicRfSendPacket()`

| 函数原型 | uint8 basicRfSendPacket(uint16 destAddr,uint8 *pPayload,uint8 length)                      |
| ---- | ------------------------------------------------------------------------------------------ |
| 函数功能 | 将待发送数据缓冲区中指定长度的数据发送给目标地址的节点                                                                |
| 返回值  | `SUCCESS`: 数据发送成功<br>`FAILED`: 数据发送失败                                                      |
| 参数说明 | `destAddr`: 发送的目标地址<br>`pPayload`: 待发送数据的缓冲区指针<br>`length`: 待发送数据的长度                       |
| 所在文件 | `basic_rf.c`                                                                               |
| 设计参考 | uint8 send_buf[16];<br>uint8 status;<br>status = basicRfSendPcaket(SEND_ADDR,send_buf,16); |

- `basicRfPcaketlsReady()` 和 `basicRfReceive()`

| 函数原型 | uint8 basicRfReceive(uint8 *pRxData,uint16 len,int6 *pRssi)                                                                                               |
| ---- | --------------------------------------------------------------------------------------------------------------------------------------------------------- |
| 函数功能 | 在已经接收到的无线数据中，读取指定长度的数据到数据接收缓冲区                                                                                                                            |
| 返回值  | 实际读到的数据长度                                                                                                                                                 |
| 参数说明 | `pRxData`: 存放读取数据的缓冲区指针 <br>`len`: 需要读取数据的长度<br>`pRssi`: 保持上一次数据接收包信息变量位置，一般填NULL                                                                         |
| 所在文件 | `basic_rf.c`                                                                                                                                              |
| 设计参考 | 先判断是否接收到无线数据，再调用该函数去读取<br>uint8 recv_buf[16];<br>uint16 len;<br>if(basicRfPacketIsReady() == TRUE)<br>{<br>  len = basicRfReceive(recv_buf,16,NULL);<br>} |
#### 实现代码

```c
#include "hal_defs.h"
#include "hal_cc8051.h"
#include "hal_int.h"
#include "hal_mcu.h"
#include "hal_board.h"
#include "hal_led.h"
#include "hal_rf.h"
#include "basic_rf.h"
#include "hal_uart.h" 
#include <stdio.h>
#include <string.h>
#include <stdarg.h>

#include "basic_rf.h"

#define My_Addr 0x000A
#define Send_Addr 0x000B
#define PAN_ID 0x1234
#define CHANNEL 15

unsigned char tx = 0x02;
unsigned char rx = 0;

static basicRfCfg_t rf;

void Init_Port()
{
  halBoardInit();
  halLedInit();
  basicRfReceiveOn();
}

void rf_init()
{
  rf.panId = PAN_ID;
  rf.channel = CHANNEL;
  rf.myAddr = My_Addr;
  rf.ackRequest = TRUE;
  
  if (basicRfInit(&rf) == FAILED)
  {
    while(1)
    {
      halLedSet(2);
      
    }
  }
  basicRfReceiveOn();
}

void rf_recv_data()
{
  if(basicRfPacketIsReady())
  {
    if(basicRfReceive(&rx,20,NULL))
    {
      if(rx == 0x02){
        halLedSet(1)
      }
    }
  }
}

/********************MAIN************************/
void main(void)
{
    Init_Port();
    rf_init();
    basicRfSendPacket(Send_Addr,&tx,sizeof(tx));
    while(1)
    {
      halLedSet(2);
      halLedSet(3);
      rf_recv_data();
    }
}
```
