[[CC2530寄存器]]
```c
#include <ioCC2530.h>

#define LED1 P1_0        
#define LED2 P1_1
#define SW1 P1_2
#define unint unsigned int

void init_gpio() 
{
    P1SEL &= ~0x07; //把0 1 2配置为1（GPIO）
    P1DIR |= 0x03; //把0 1配置为0（输出）
    P1DIR &= ~0x04; //把2配置为1（输入） 

    P1 = 0x00;
    
    // 设置端口1为上下拉功能
    P1INP &=~ 0x04;
    // 设置端口1输入模式为上拉
    P2INP &=~ 0x40;
}

void delay(unint z)
{
    for (unint i = 0; i < z; i++)
    {
        for (unint j = 0; j < 500; j++);
    }
}

void main(void) {
    init_gpio();
    while (1) 
    {
      if (SW1 == 0){
        LED1 = 1;
      }else {
        LED1 = 0;
        LED2 = 1;
      }
    }
}
```

### Init_GPIO函数
#### - 把`P1.0``P1.1``P1.2`的`SEL`清零
```c
P1SEL &=~ 0x07; //0x07对应0000 0111  ~0x07对应1111 1000
```

#### - 把P1.0P1.1的`DIR`配置为输出(1)
```c
P1DIR |= 0x03; //0x03对应0000 0011
```

#### - 把P1.2的DIR配置为输入(0)
```c
P1DIR &=~ 0x04; //0x04对应0000 0100 ~0x04对应1111 1011
```

#### - 把P1寄存器置于低位(0)
```c
P1 = 0x00;
```

#### - 设置端口P1.1为上下拉功能
```c
P1INP &=~ 0x04;
```


#### - 设置端口P1.1输入模式为上拉
```c
P2INP &=~ 0x40;
```
