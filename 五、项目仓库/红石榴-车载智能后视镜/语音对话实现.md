### 麦克风

#### 声音采集
```c
int cp_audio_capture_start(void)
{
    union audio_req request;
    if (!s_initialized && cp_audio_capture_init() != 0) return -1;
    if (s_running) return 0;
    s_encoder = server_open("audio_server", "enc");
    if (s_encoder == NULL) return -1;
    memset(&request, 0, sizeof(request));
    request.enc.cmd = AUDIO_ENC_OPEN;
    request.enc.channel = BOARD_AUDIO_CAPTURE_CHANNELS;
#if BOARD_AUDIO_CAPTURE_CHANNELS == 1U
    request.enc.channel_bit_map = BIT(JIELI_MIC0_LOGICAL_CHANNEL);
#else
    request.enc.channel_bit_map = BIT(JIELI_MIC0_LOGICAL_CHANNEL) |
                                  BIT(JIELI_MIC2_LOGICAL_CHANNEL);
#endif
    request.enc.volume = CONFIG_AISP_MIC_ADC_GAIN;
    request.enc.sample_rate = CP_AUDIO_SAMPLE_RATE;
    request.enc.format = "pcm";
    request.enc.sample_source = "mic";
    request.enc.frame_size = CP_AUDIO_FRAME_SAMPLES *
                             BOARD_AUDIO_CAPTURE_CHANNELS * sizeof(int16_t);
    request.enc.output_buf_len = request.enc.frame_size * 3U;
    request.enc.vfs_ops = &s_capture_vfs;
    request.enc.file = (FILE *)&s_capture_cbuf;
    if (server_request(s_encoder, AUDIO_REQ_ENC, &request) != 0) {
        server_close(s_encoder);
        s_encoder = NULL;
        return -1;
    }
    s_running = 1U;
    return 0;
}
```

`cp_audio_capture_start()` 打开杰理 SDK 的 `audio_server`，设置 `sample_source="mic"`、`format="pcm"`、采样率 `16000`。可以理解为告诉芯片：“开始采集麦克风，并把声音转换成数字样本交给我。”这里虽然使用 `"enc"` 接口，但输出配置为 PCM，没有压成 MP3。


SDK采集到数据后，调用：

```c
static int capture_fwrite(void *file, void *data, u32 len)
{
    u32 written = cbuf_write((cbuffer_t *)file, data, len);
    if (written != len) {
        s_overruns++;
        cbuf_clear(&s_capture_cbuf);
        written = cbuf_write(&s_capture_cbuf, data, len);
    }
    return written;
}
```

#### 音频前端处理（双麦算法），输出单声道

```c
int cp_audio_frontend_process(const cp_audio_capture_frame_t *input,
                              uint32_t now_ms)
{

    cp_audio_processed_frame_t output;
    float echo_gain = 0.0f;
    float ref_energy = 0.0f;
    float correlation = 0.0f;
    float mic_energy[CP_AUDIO_MIC_CHANNELS] = {0.0f, 0.0f};
    float mic_rms[CP_AUDIO_MIC_CHANNELS];
    float mic_score[CP_AUDIO_MIC_CHANNELS];
    float sum_square = 0.0f;
    float rms;
    float gate;
    float primary_weight;
    uint8_t previous_mic;
    uint8_t switched;
    uint8_t calibrating;
    uint8_t mic_voice[CP_AUDIO_MIC_CHANNELS];
    uint16_t i;
    uint8_t active_bands = 0;
    if (input == NULL || input->sample_count == 0U ||
        input->sample_count > CP_AUDIO_FRAME_SAMPLES) return -1;
    memset(&output, 0, sizeof(output));
    output.sample_count = input->sample_count;
    for (i = 0; i < input->sample_count; i++) {
        float left = dc_block((float)input->mic0[i], 0U);
        float right = dc_block((float)input->mic2[i], 1U);
        s_mic_pcm[CP_AUDIO_MIC0_INDEX][i] = clamp_pcm(left);
        s_mic_pcm[CP_AUDIO_MIC2_INDEX][i] = clamp_pcm(right);
        mic_energy[CP_AUDIO_MIC0_INDEX] += left * left;
        mic_energy[CP_AUDIO_MIC2_INDEX] += right * right;
        if (input->mic0[i] >= CLIP_LEVEL || input->mic0[i] <= -CLIP_LEVEL) {
            output.metadata.mic_clipped_samples[CP_AUDIO_MIC0_INDEX]++;
        }
        if (input->mic2[i] >= CLIP_LEVEL || input->mic2[i] <= -CLIP_LEVEL) {
            output.metadata.mic_clipped_samples[CP_AUDIO_MIC2_INDEX]++;
        }
    }
    calibrating = s_calibration_frames_left != 0U;
    if (calibrating) s_calibration_frames_left--;
    for (i = 0; i < CP_AUDIO_MIC_CHANNELS; i++) {
        mic_rms[i] = rms_from_energy(mic_energy[i], input->sample_count);
        mic_score[i] = mic_rms[i] / s_mic_noise_floor[i];
        mic_voice[i] = (!calibrating &&
                        mic_rms[i] >= s_config.vad_threshold &&
                        mic_score[i] >= 2.5f) ? 1U : 0U;
        update_noise_floor((uint8_t)i, mic_rms[i], calibrating, mic_voice[i]);
        output.metadata.mic_rms[i] = mic_rms[i];
        output.metadata.mic_noise_floor[i] = s_mic_noise_floor[i];
        output.metadata.mic_snr_db[i] = snr_db(mic_rms[i],
                                               s_mic_noise_floor[i]);
    }
    switched = select_microphone(mic_rms, mic_score, &previous_mic);
    output.metadata.selected_mic = s_selected_mic;
    output.metadata.mic_switched = switched;
    output.metadata.noise_calibrating = calibrating;
    primary_weight = 1.0f - s_config.secondary_mic_mix;
  
    for (i = 0; i < input->sample_count; i++) {
        float sample;
        if (switched) {
            float fade = (float)(i + 1U) / (float)input->sample_count;
            sample = (float)s_mic_pcm[previous_mic][i] * (1.0f - fade) +
                     (float)s_mic_pcm[s_selected_mic][i] * fade;
        } else {
            sample = (float)s_mic_pcm[s_selected_mic][i] * primary_weight +
                     (float)s_mic_pcm[s_selected_mic ^ 1U][i] *
                     s_config.secondary_mic_mix;
        }
        output.pcm[i] = clamp_pcm(sample);
        if (input->has_reference) {
            float reference = (float)input->reference[i];
            correlation += sample * reference;
            ref_energy += reference * reference;
        }
    }

    if (ref_energy > 1.0f) {

        echo_gain = correlation / ref_energy;
        if (echo_gain < 0.0f) echo_gain = 0.0f;
        if (echo_gain > s_config.echo_gain_limit) {
            echo_gain = s_config.echo_gain_limit;
        }
        if (echo_gain > 0.01f) output.metadata.echo_cancelled = 1U;
    }

    for (i = 0; i < input->sample_count; i++) {
        float sample = (float)output.pcm[i];

        if (input->has_reference) sample -= echo_gain * input->reference[i];
        output.pcm[i] = clamp_pcm(sample);
        sum_square += sample * sample;
    }

    rms = sqrtf(sum_square / (float)input->sample_count);
    output.metadata.noise_floor =
        s_mic_noise_floor[s_selected_mic] * primary_weight +
        s_mic_noise_floor[s_selected_mic ^ 1U] * s_config.secondary_mic_mix;
    gate = output.metadata.noise_floor * s_config.noise_gate_ratio;

    if (gate < s_config.vad_threshold * 0.5f) gate = s_config.vad_threshold * 0.5f;

    if (rms < gate) {
        for (i = 0; i < input->sample_count; i++) output.pcm[i] /= 8;
    }

    if (!calibrating && (mic_voice[CP_AUDIO_MIC0_INDEX] ||
                         mic_voice[CP_AUDIO_MIC2_INDEX]) &&
        rms >= s_config.vad_threshold * 0.5f) {
        output.metadata.valid_voice = 1U;
        s_vad_hangover_left = s_vad_hangover_frames;
    } else if (s_vad_hangover_left != 0U) {
        s_vad_hangover_left--;
    }

    output.metadata.vad = (output.metadata.valid_voice ||
                           s_vad_hangover_left != 0U) ? 1U : 0U;
    output.metadata.timestamp_ms = now_ms;
    output.metadata.rms = rms;
    output.metadata.dbfs = rms > 0.0f ?
                           20.0f * log10f(rms / PCM_FULL_SCALE) : -96.0f;
    calculate_bands(output.pcm, output.sample_count, output.metadata.bands);

    for (i = 0; i < CP_AUDIO_BAND_COUNT; i++) {
        if (output.metadata.bands[i] > 0.035f) active_bands++;
    }

    output.metadata.music = (output.metadata.vad && active_bands >= 5U) ? 1U : 0U;
    output.metadata.sequence = ++s_sequence;
  
    if (output.metadata.vad != s_last_vad) {
        cp_event_t event = {CP_EVENT_AUDIO_ACTIVITY, now_ms,
                            output.metadata.vad, output.metadata.music};
        cp_event_post(&event);
        s_last_vad = output.metadata.vad;
    }

    if (os_mutex_lock(s_lock, 20U) != 0) return -1;
    s_latest = output;
    os_mutex_unlock(s_lock);
    return 0;
}
```

#### PCM上传云端链路



|消息|发送时机|云端如何理解|
|---|---|---|
|`dialog_start`|开始倾听|新建一轮对话，记录 `session_id`|
|`dialog_pcm`|每次上传音频帧|这一轮对话的下一段声音|
|`dialog_end`|判断一句话结束|音频已发完，可以完成识别和回答|
|`dialog_cancel`|取消或超时|放弃这一轮对话|


`write_audio()`函数负责把`int16_t`数组交给如下函数：
```c
int cp_comm_service_publish_dialog(const char *type, const char *session_id,
                                   const uint8_t *data, uint16_t length,
                                   uint32_t timestamp_ms)
```
再由此函数把每个字节编码成两个十六进制字符，例如字节 `0x12 0xAB` 变成 `"12ab"`，再组装成这样的 JSON：
```json
{"device_id":"设备IMEI",
"type":"dialog_pcm",
"timestamp":12345,
"seq":20,
 "body":{
 "session_id":"本轮对话编号","data_hex":"12ab..."
 }
 }
```

- `data_hex` 是声音数据的文本表示，**不是加密，也不是压缩，大小会翻倍**。
- `session_id` 用来区分不同轮对话，
- `seq` 是消息序号；这里的时间戳来自设备运行计时，不是日历时间。


真正发送的位置是 comm_transport_jieli.c (line 102) 的 `mqtt_publish()`：它调用 SDK 的 `MQTTPublish()`，发布到 **`device/设备IMEI/up`**。`mqtt_connect()` 负责建立网络连接、登录 MQTT broker，并订阅下行主题。MQTT broker 可以理解为消息中转站，云端业务服务订阅上行主题后才能拿到音频。
